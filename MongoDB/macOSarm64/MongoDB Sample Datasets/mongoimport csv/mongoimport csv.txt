# mongoimport
# Download .csv for mongoimport Dataset for Use on Local Machine
% curl https://data.nasa.gov/api/views/9kcy-zwvn/rows.csv -o eva.csv  
# Or
% wget https://data.nasa.gov/api/views/9kcy-zwvn/rows.csv -O eva.csv

[
% brew services start mongodb-community
]
% mongoimport --headerline --file eva.csv --type csv --db nasa --collection eva
% mongosh
